? কীভাবে ২ মিনিটে গুগল শিট যুক্ত করবেন (নির্দেশিকা):

  1. আপনার গুগল ড্রাইভে একটি নতুন Google Sheet খুলুন (নাম দিন: বাহাদুরপুর মুজাহিদ মাহফিল হিসাব)।
  2. শিটের উপরের মেনু থেকে Extensions → Apps Script এ ক্লিক করুন।
  3. সেখানে থাকা আগের কোড মুছে নিচের "Apps Script কোড কপি করুন" বাটনে চাপ দিয়ে কোডটি পেস্ট করুন এবং সেভ (Ctrl+S) করুন।
  4. উপরে ডানপাশে Deploy → New deployment ক্লিক করুন → Type নির্বাচন করুন Web app।
  5. Execute as: Me এবং Who has access: Anyone দিয়ে Deploy করুন।
  6. প্রাপ্ত Web App URL-টি কপি করে উপরের বক্সে পেস্ট করুন এবং "কানেক্ট ও লাইভ সিঙ্ক করুন" বাটনে চাপ দিন।
  7. এই একই URL-টি কমিটির অন্যান্য দায়িত্বশীলদের মোবাইলে দিয়ে দিলে সবাই একসাথে লাইভ হিসাব দেখতে ও আপডেট করতে পারবেন!

// Google Apps Script for Bahadurpur Union Mujahid Committee
function doGet(e) {
  return handleRequest('GET', null);
}

function doPost(e) {
  var payload = JSON.parse(e.postData.contents);
  return handleRequest('POST', payload);
}

function handleRequest(method, payload) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var mSheet = getOrCreateSheet(ss, 'Mujahid_List', ['id', 'name', 'phone']);
    var tSheet = getOrCreateSheet(ss, 'Transactions', ['id', 'type', 'memberId', 'category', 'amount', 'date', 'note']);
    var cSheet = getOrCreateSheet(ss, 'Categories', ['type', 'name']);

    if (method === 'POST' && payload) {
      if (payload.action === 'syncAll') {
        writeSheetData(mSheet, ['id', 'name', 'phone'], payload.data.members || []);
        writeSheetData(tSheet, ['id', 'type', 'memberId', 'category', 'amount', 'date', 'note'], payload.data.transactions || []);
        var catRows = [];
        (payload.data.categories.income || []).forEach(function(c) { catRows.push({ type: 'income', name: c }); });
        (payload.data.categories.expense || []).forEach(function(c) { catRows.push({ type: 'expense', name: c }); });
        writeSheetData(cSheet, ['type', 'name'], catRows);
      } else if (payload.action === 'addTransaction') {
        var tx = payload.item;
        tSheet.appendRow([tx.id, tx.type, tx.memberId, tx.category, tx.amount, tx.date, tx.note || '']);
      } else if (payload.action === 'deleteTransaction') {
        deleteRowById(tSheet, payload.id);
      } else if (payload.action === 'addMember') {
        var m = payload.item;
        mSheet.appendRow([m.id, m.name, "'" + m.phone]);
      } else if (payload.action === 'updateMember') {
        updateMemberRow(mSheet, payload.item);
      } else if (payload.action === 'deleteMember') {
        deleteRowById(mSheet, payload.id);
      } else if (payload.action === 'addCategory') {
        cSheet.appendRow([payload.catType, payload.name]);
      }
    }

    var members = readSheetObjects(mSheet).map(function(r) {
      return { id: String(r.id), name: String(r.name), phone: String(r.phone).replace(/^'/, '') };
    });
    var transactions = readSheetObjects(tSheet).map(function(r) {
      return {
        id: String(r.id),
        type: String(r.type),
        memberId: String(r.memberId),
        category: String(r.category),
        amount: Number(r.amount) || 0,
        date: formatSheetDate(r.date),
        note: String(r.note || '')
      };
    });
    var catObjs = readSheetObjects(cSheet);
    var categories = { income: [], expense: [] };
    catObjs.forEach(function(c) {
      if (c.type === 'income' && c.name) categories.income.push(String(c.name));
      if (c.type === 'expense' && c.name) categories.expense.push(String(c.name));
    });

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      members: members,
      transactions: transactions,
      categories: categories
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#064e3b').setFontColor('#ffffff');
  }
  return sheet;
}

function readSheetObjects(sheet) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    if (!data[i][0] && !data[i][1]) continue;
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    rows.push(obj);
  }
  return rows;
}

function writeSheetData(sheet, headers, items) {
  sheet.clearContents();
  sheet.appendRow(headers);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#064e3b').setFontColor('#ffffff');
  if (items.length === 0) return;
  var rows = items.map(function(item) {
    return headers.map(function(h) {
      return h === 'phone' ? "'" + item[h] : (item[h] !== undefined ? item[h] : '');
    });
  });
  sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
}

function deleteRowById(sheet, id) {
  var data = sheet.getDataRange().getValues();
  for (var i = data.length - 1; i >= 1; i--) {
    if (String(data[i][0]) === String(id)) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
}

function updateMemberRow(sheet, member) {
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(member.id)) {
      sheet.getRange(i + 1, 2, 1, 2).setValues([[member.name, "'" + member.phone]]);
      break;
    }
  }
}

function formatSheetDate(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(val || '');
}
