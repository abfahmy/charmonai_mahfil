function doGet(e) {
  return handleRequest('GET', null);
}

function doPost(e) {
  var payload = e && e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {};
  return handleRequest('POST', payload);
}

function handleRequest(method, payload) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var mSheet = ensureSheet(ss, 'Members', ['id', 'name', 'phone']);
    var tSheet = ensureSheet(ss, 'Transactions', ['id', 'type', 'memberId', 'category', 'amount', 'date', 'note']);
    var cSheet = ensureSheet(ss, 'Categories', ['name']);

    if (method === 'POST' && payload) {
      if (payload.action === 'syncAll') {
        writeObjects(mSheet, ['id', 'name', 'phone'], (payload.data && payload.data.members) || []);
        writeObjects(tSheet, ['id', 'type', 'memberId', 'category', 'amount', 'date', 'note'], (payload.data && payload.data.transactions) || []);
        var catList = ((payload.data && payload.data.categories) || []).map(function(c) { return { name: c }; });
        writeObjects(cSheet, ['name'], catList);
      } else if (payload.action === 'addTransaction' && payload.item) {
        var tx = payload.item;
        tSheet.appendRow([tx.id, tx.type, tx.memberId, tx.category, tx.amount, tx.date, tx.note || '']);
      } else if (payload.action === 'editTransaction' && payload.item) {
        updateRowById(tSheet, payload.item.id, ['type', 'memberId', 'category', 'amount', 'date', 'note'], payload.item);
      } else if (payload.action === 'deleteTransaction' && payload.id) {
        deleteRowById(tSheet, payload.id);
      } else if (payload.action === 'addMember' && payload.item) {
        var m = payload.item;
        mSheet.appendRow([m.id, m.name, "'" + m.phone]);
      } else if (payload.action === 'updateMember' && payload.item) {
        updateRowById(mSheet, payload.item.id, ['name', 'phone'], { name: payload.item.name, phone: "'" + payload.item.phone });
      } else if (payload.action === 'deleteMember' && payload.id) {
        deleteRowById(mSheet, payload.id);
      } else if (payload.action === 'addCategory' && payload.name) {
        cSheet.appendRow([payload.name]);
      }
    }

    var members = readObjects(mSheet).map(function(r) {
      return {
        id: String(r.id || ''),
        name: String(r.name || ''),
        phone: String(r.phone || '').replace(/^'/, '')
      };
    }).filter(function(m) { return m.id && m.name; });

    var transactions = readObjects(tSheet).map(function(r) {
      return {
        id: String(r.id || ''),
        type: r.type === 'expense' ? 'expense' : 'income',
        memberId: String(r.memberId || 'general'),
        category: String(r.category || 'অন্যান্য'),
        amount: Number(r.amount) || 0,
        date: normalizeDate(r.date),
        note: String(r.note || '')
      };
    }).filter(function(t) { return t.id && t.amount > 0; });

    var categories = readObjects(cSheet).map(function(c) {
      return String(c.name || '').trim();
    }).filter(function(c) { return c.length > 0; });

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      members: members,
      transactions: transactions,
      categories: categories
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function ensureSheet(ss, name, headers) {
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow(headers);
    sh.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#064e3b').setFontColor('#ffffff');
  }
  return sh;
}

function readObjects(sh) {
  var values = sh.getDataRange().getValues();
  if (values.length <= 1) return [];
  var headers = values[0];
  var out = [];
  for (var i = 1; i < values.length; i++) {
    var obj = {};
    var hasVal = false;
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = values[i][j];
      if (values[i][j] !== '') hasVal = true;
    }
    if (hasVal) out.push(obj);
  }
  return out;
}

function writeObjects(sh, headers, list) {
  sh.clearContents();
  sh.appendRow(headers);
  sh.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#064e3b').setFontColor('#ffffff');
  if (!list || list.length === 0) return;
  var rows = list.map(function(item) {
    return headers.map(function(h) {
      if (h === 'phone') return "'" + String(item[h] || '');
      return item[h] !== undefined ? item[h] : '';
    });
  });
  sh.getRange(2, 1, rows.length, headers.length).setValues(rows);
}

function updateRowById(sh, id, fields, dataObj) {
  var values = sh.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][0]) === String(id)) {
      var rowVals = fields.map(function(f) { return dataObj[f] !== undefined ? dataObj[f] : ''; });
      sh.getRange(i + 1, 2, 1, rowVals.length).setValues([rowVals]);
      break;
    }
  }
}

function deleteRowById(sh, id) {
  var values = sh.getDataRange().getValues();
  for (var i = values.length - 1; i >= 1; i--) {
    if (String(values[i][0]) === String(id)) {
      sh.deleteRow(i + 1);
      break;
    }
  }
}

function normalizeDate(val) {
  if (Object.prototype.toString.call(val) === '[object Date]' && !isNaN(val.getTime())) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(val || '');
}
